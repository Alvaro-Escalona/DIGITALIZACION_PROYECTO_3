Transversor PDF - Document Translation Tool

This application allows users to translate PDF documents into multiple languages while maintaining a professional output format in both PDF and DOCX.

Main features:

  • PDF text extraction using PyMuPDF (fitz).
  • Multi-language translation via Google Translator.
  • Selection of specific pages for processing.
  • Export to Word (.docx) and PDF formats.

344    def hilo_traduccion(self, config):
345        """
346        Core translation engine executed in a background thread.
347        Handles text extraction, chunking, translation, and file generation.
348
349        Args:
350            config (dict): Configuration containing language, format, and structure settings.
351        """
352        try:
353            self._set_progress_ui(0)
354            translator = GoogleTranslator(source='auto', target=config["idioma_destino"])
355            archivos = list(self.cola_archivos.items())
356            total_bloques = 0
357            estructura_archivos = []
358
359            # Preparation Phase: Calculate total work
360            for p_in, paginas in archivos:
361                if not paginas: continue
362                reader = PdfReader(p_in)
363                doc_fitz = fitz.open(p_in)
364                paginas_origen = []
365                for idx in sorted(set(paginas)):
366                    parrafos = self.extraer_parrafos_pagina(doc_fitz, idx)
367                    if not parrafos:
368                        t = reader.pages[idx].extract_text()
369                        parrafos = self.extraer_parrafos(t) if t else []
370                    paginas_origen.append(parrafos)
371                    for p in parrafos:
372                        total_bloques += max(1, len(self.construir_bloques_traduccion(p)))
373                estructura_archivos.append((p_in, paginas_origen))
374                doc_fitz.close()
375
376            if total_bloques <= 0: total_bloques = 1
377            bloques_completados = 0
378
379            # Processing Phase: Translate and Save
380            for i, (p_in, paginas_origen) in enumerate(estructura_archivos):
381                self._set_status_ui(f"Translating {i+1}/{len(estructura_archivos)}...", "orange")
382                paginas_traducidas = []
383                
384                def avanzar_progreso():
385                    nonlocal bloques_completados
386                    bloques_completados += 1
387                    self._set_progress_ui(bloques_completados / total_bloques)
388
389                for parrafos in paginas_origen:
390                    parrafos_traducidos = [self.traducir_texto(translator, p, progreso_callback=avanzar_progreso) for p in parrafos]
391                    paginas_traducidas.append(parrafos_traducidos)
392
393                ext = ".pdf" if config["formato"] == "PDF" else ".docx"
394                p_out = os.path.join(self.ruta_destino, f"Translated_{os.path.basename(p_in).replace('.pdf', ext)}")
395                
396                if config["formato"] == "PDF":
397                    self.guardar_pdf_estructurado(paginas_traducidas, p_out, separar_paginas=config["separar_paginas"])
398                else:
399                    self.guardar_docx(paginas_traducidas, p_out, separar_paginas=config["separar_paginas"])
400
401            self._set_progress_ui(1)
402            self._set_status_ui("COMPLETED!", "green")
403            self.after(0, lambda: messagebox.showinfo("Success", "Files processed correctly."))
404        except Exception as e:
405            self._set_status_ui("Translation Error", "red")
406            self.after(0, lambda: messagebox.showerror("Error", f"Failed:\n{e}"))
407        finally:
408            self.after(0, self._cerrar_ventana_progreso_ui)
409            self.after(0, lambda: self.btn_traducir.configure(state="normal"))

411    def guardar_pdf_estructurado(self, paginas_texto, path, separar_paginas=True):
412        """
413        Generates a professional PDF with basic structure and text wrapping.
414
415        Args:
416            paginas_texto (list): Nested list of translated paragraphs by page.
417            path (str): Destination file path.
418            separar_paginas (bool): Whether to respect original page breaks.
419        """
420        c = canvas.Canvas(path, pagesize=A4)
421        text_obj = c.beginText(50, 800)
422        text_obj.setFont("Helvetica", 10)
423        pagina_vacia = True
424
425        for i, parrafos_pagina in enumerate(paginas_texto):
426            if parrafos_pagina:
427                for parrafo in parrafos_pagina:
428                    texto = parrafo.strip()
429                    if not texto: continue
430                    lineas = textwrap.wrap(texto, width=95)
431                    for linea in lineas:
432                        if text_obj.getY() < 50:
433                            c.drawText(text_obj); c.showPage()
434                            text_obj = c.beginText(50, 800); text_obj.setFont("Helvetica", 10)
435                            pagina_vacia = True
436                        text_obj.textLine(linea)
437                        pagina_vacia = False
438                    text_obj.textLine("") 
439
440            if separar_paginas and i < len(paginas_texto) - 1:
441                if not pagina_vacia:
442                    c.drawText(text_obj); c.showPage()
443                    text_obj = c.beginText(50, 800); text_obj.setFont("Helvetica", 10)
444                    pagina_vacia = True
445
446        c.drawText(text_obj); c.save()

448    def construir_bloques_traduccion(self, texto, tam_bloque=1800):
449        """
450        Splits a paragraph into smaller chunks to respect API character limits.
451
452        Args:
453            texto (str): The full paragraph string.
454            tam_bloque (int): Maximum character count per chunk.
455
456        Returns:
457            list: List of text chunks.
458        """
459        contenido = texto.strip()
460        if not contenido: return []
461        oraciones = self.dividir_en_oraciones(contenido)
462        bloques, actual = [], ""
463        for oracion in oraciones:
464            candidato = (actual + " " + oracion).strip() if actual else oracion
465            if len(candidato) <= tam_bloque: actual = candidato
466            else:
467                if actual: bloques.append(actual)
468                actual = oracion
469        if actual: bloques.append(actual)
470        return bloques

472    def traducir_texto(self, translator, texto, tam_bloque=1800, progreso_callback=None):
473        """
474        Orchestrates the translation of a single paragraph through chunks.
475
476        Args:
477            translator: The GoogleTranslator instance.
478            texto (str): Paragraph to translate.
479            tam_bloque (int): Max chunk size.
480            progreso_callback (callable): Function to call after each successful chunk translation.
481
482        Returns:
483            str: The fully translated paragraph.
484        """
485        if not texto.strip(): return ""
486        prefijo = ""
487        contenido = texto.strip()
488        if contenido.startswith(("- ", "• ", "* ")):

512    def extraer_parrafos_pagina(self, doc_fitz, page_index):
513        """
514        Intelligent text extraction using visual blocks to identify paragraphs.
515
516        Args:
517            doc_fitz: PyMuPDF document instance.
518            page_index (int): Page number to extract.
519
520        Returns:
521            list: List of identified paragraphs.
522        """
523        page = doc_fitz[page_index]
524        blocks = sorted(page.get_text("dict").get("blocks", []), key=lambda b: (round(b.get("bbox", [0, 0])[1], 1), round(b.get("bbox", [0, 0])[0], 1)))
525        parrafos = []
526        for bloque in blocks:
527            if bloque.get("type") != 0: continue
528            actual = ""
529            for linea in bloque.get("lines", []):
530                t_linea = " ".join([s.get("text", "").strip() for s in linea.get("spans", []) if s.get("text", "").strip()]).strip()
531                if not t_linea: continue
532                if not actual: actual = t_linea
533                else: actual = (actual[:-1] + t_linea) if actual.endswith("-") else (actual + " " + t_linea)
534            if actual.strip(): parrafos.append(actual.strip())
535        return parrafos

349
350        Args:

352        """

356            archivos = list(self.cola_archivos.items())
357            total_bloques = 0

372                    for p in parrafos:

374                estructura_archivos.append((p_in, paginas_origen))
